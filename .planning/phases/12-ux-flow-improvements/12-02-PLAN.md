- To: export const Editor = forwardRef<EditorHandle, EditorProps>(({ ... }, ref) => { ... })

3. Create internal CodeMirror ref:

```typescript
const cmRef = useRef<ReactCodeMirrorRef>(null);
```

4. Expose focus method via useImperativeHandle:

```typescript
useImperativeHandle(ref, () => ({
  focus: () => {
    cmRef.current?.view?.focus();
  },
}));
```

5. Pass cmRef to CodeMirror component:

```typescript
<CodeMirror
  ref={cmRef}
  // ... existing props
/>
```
